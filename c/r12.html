题1学生成绩

代表题 · 教材第9章习题5（川师本科卷 2016-17、2018-19 都考过这个题型）

有 10 个学生，每个学生的数据包括学号、姓名、3 门课程的成绩，从键盘输入 10 个学生数据，要求输出 3 门课程总平均成绩，以及最高分的学生的数据（包括学号、姓名、3 门课程成绩、平均分数）。

要求

「3 门课程总平均成绩」有两种理解，这里两个都写：

course_avg(s, n, avg)：avg[0]、avg[1]、avg[2] 分别是三门课各自的平均分。

best_student(s, n)：返回平均分最高的学生的下标（并列取前面那个）。

对应讲义　③

易错按列求平均；最高者记下标，比较平均分。

#include <stdio.h>

struct Student {
    int num;
    char name[20];
    double score[3];
};

void course_avg(struct Student s[], int n, double avg[3])
{
    int i, k;
    for (k = 0; k < 3; k++) {
        double sum = 0;
        for (i = 0; i < n; i++) sum += s[i].score[k];
        avg[k] = sum / n;
    }
}

int best_student(struct Student s[], int n)
{
    int i, best = 0;
    double a, b;
    for (i = 1; i < n; i++) {
        a = (s[i].score[0] + s[i].score[1] + s[i].score[2]) / 3;
        b = (s[best].score[0] + s[best].score[1] + s[best].score[2]) / 3;
        if (a > b) best = i;
    }
    return best;
}

#define N 10

int main(void)
{
    struct Student s[N];
    double avg[3];
    int i, b;
    for (i = 0; i < N; i++)
        scanf("%d%s%lf%lf%lf", &s[i].num, s[i].name, &s[i].score[0], &s[i].score[1], &s[i].score[2]);
    course_avg(s, N, avg);
    printf("三门课的平均分：%.2f %.2f %.2f\n", avg[0], avg[1], avg[2]);
    b = best_student(s, N);
    printf("最高分：%d %s %.1f %.1f %.1f 平均 %.2f\n", s[b].num, s[b].name,
           s[b].score[0], s[b].score[1], s[b].score[2],
           (s[b].score[0] + s[b].score[1] + s[b].score[2]) / 3);
    return 0;
}

102 Wang 85 75 95
103 Zhang 90 92 94
104 Liu 60 70 80
105 Chen 88 66 77
106 Yang 70 80 90
107 Zhao 95 85 75
108 Huang 66 77 88

题2结构体日期

变式 · 教材第9章习题1（和环01 题5「当年第几天」是同一个算法）

定义一个结构体变量（包括年、月、日）。计算该日在本年中是第几天，注意闰年问题。

要求

day_of_year2(d) 返回 d 这一天是这一年的第几天。用 d.year、d.month、d.day 取成员。

对应讲义　① 环01 ⑦

易错前 month-1 个月的天数加 day；闰年 2 月 29 天。

#include <stdio.h>

struct Date {
    int year, month, day;
};

int day_of_year2(struct Date d)
{
    int days[13] = {0, 31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    int i, total = d.day;
    if ((d.year % 4 == 0 && d.year % 100 != 0) || d.year % 400 == 0) days[2] = 29;
    for (i = 1; i < d.month; i++) total += days[i];
    return total;
}

int main(void)
{
    struct Date d;
    scanf("%d%d%d", &d.year, &d.month, &d.day);
    printf("第 %d 天\n", day_of_year2(d));
    return 0;
}

输入

2024 3 1

运行结果

第 61 天

题4成绩排序

综合 · 圣才配套题库 第9章设计题1

学生的记录由学号和成绩组成，N 名学生的数据已在主函数中放入结构体数组 s 中，请编写函数 fun，它的功能是：按分数的高低排列学生的记录，高分在前。

（原题框架：for(i=0;i<N;i++) for(j=i+1;j<N;j++) { 请填写 }，fun 声明成 int 却没有 return，

那是原题框架的毛病，这里写成 void）

要求

fun(a, n) 按 s 从高到低排序，交换整条记录。

对应讲义　④

易错交换整条记录 tmp = a[i]。

#include <stdio.h>

typedef struct {
    char num[10];
    int s;
} STREC;

void fun(STREC a[], int n)
{
    STREC tmp;
    int i, j;
    for (i = 0; i < n; i++)
        for (j = i + 1; j < n; j++)
            if (a[i].s < a[j].s) { tmp = a[i]; a[i] = a[j]; a[j] = tmp; }
}

#define N 8

int main(void)
{
    STREC s[N] = {{"GA005", 85}, {"GA003", 76}, {"GA002", 69}, {"GA004", 85},
                  {"GA001", 91}, {"GA007", 72}, {"GA008", 64}, {"GA006", 87}};
    int i;
    fun(s, N);
    for (i = 0; i < N; i++) printf("%s %d\n", s[i].num, s[i].s);
    return 0;
}

运行结果

GA006 87
GA005 85
GA004 85
GA003 76

题5链表删除

限时 · 教材第9章习题7 · 限时 15 分钟

在第 9 章例 9.9 和例 9.10 的基础上，写一个函数 del，用来删除动态链表中指定的结点。

要求

del(head, num) 删除链表里第一个 num 等于给定值的结点，返回删除后的链表头指针；没找到就原样返回。被删的结点用 free 释放（结点都是 malloc 来的）。

  • 思路p 找结点，pre 跟在 p 后面；删的是第一个结点时头指针要变。

对应讲义　⑧

易错pre 跟着 p；删头结点要改 head；free 被删的结点。

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int num;
    struct Node *next;
};

struct Node *del(struct Node *head, int num)
{
    struct Node *p = head, *pre = NULL;
    while (p != NULL && p->num != num) { pre = p; p = p->next; }
    if (p == NULL) return head;            /* 没找到 */
    if (pre == NULL) head = p->next;       /* 删的是第一个 */
    else pre->next = p->next;
    free(p);
    return head;
}

/* 建链表（教材例 9.9 的做法）：读到 0 结束 */
struct Node *creat(void)
{
    struct Node *head = NULL, *tail = NULL, *p;
    int num;
    scanf("%d", &num);
    while (num != 0) {
        p = (struct Node *)malloc(sizeof(struct Node));
        p->num = num;
        p->next = NULL;
        if (head == NULL) head = p;
        else tail->next = p;
        tail = p;
        scanf("%d", &num);
    }
    return head;
}

void print_list(struct Node *head)
{
    struct Node *p;
    for (p = head; p != NULL; p = p->next) printf("%d ", p->num);
    printf("\n");
}

int main(void)
{
    struct Node *head;
    int num;
    head = creat();
    scanf("%d", &num);                 /* 要删的学号 */
    head = del(head, num);
    print_list(head);
    return 0;
}

输入

101 102 103 104 0
103

运行结果

101 102 104

编程题怎么写

结构体题的难点不在算法（排序、找最大都是前面学过的），而在「成员怎么取」。

通用骨架

struct Stu {
    int num;
    char name[20];
    double score;
};                                     // 分号不能漏

struct Stu s[N];
for (i = 0; i < n; i++)
    scanf("%d %s %lf", &s[i].num, s[i].name, &s[i].score);

①先定义类型，再用类型定义变量

struct Stu { … }; 只是定义了一个类型，不分配内存。类型名是 struct Stu 两个词。之后 struct Stu a = {101, "Li", 88.5}; 才是变量。

取成员用点：a.num 是 101，a.name 是 Li。

题2 的日期：struct Date { int year, month, day; };，用 d.year、d.month、d.day 取，算法和环01 ⑦ 完全一样。

②输入输出：一个成员一个成员来

scanf("%d %s %lf", &a.num, a.name, &a.score);对 scanf("%d", &a);错 结构体不能整体输入 scanf("%s", &a.name);错 字符数组成员多了 &（回顾第 1 题） printf("%d %s %.1f\n", a.num, a.name, a.score);对 输出也逐个成员

③记录表统计：找最高的，记下标

best = 0; sum = 0;
for (i = 0; i < n; i++) {
    sum += s[i].score;
    if (s[i].score > s[best].score) best = i;
}
printf("%d %s %.1f\n", s[best].num, s[best].name, s[best].score);

记下标而不是记分数：最后要输出的是这个人的整条记录（道理和环06 ② 一样）。

三条记录 {3, Wu, 70}、{1, Zh, 95}、{2, Qi, 82}：平均 82.33，最高的是 1 Zh 95.0。

④排序：比成员，换整条

if (s[j].score < s[j + 1].score) {
    t = s[j]; s[j] = s[j + 1]; s[j + 1] = t;     // t 是 struct Stu
}

同类型的结构体变量可以整体赋值，t = s[j]; 一句就复制了所有成员，连 name 数组一起。

只交换 score，分数换了、名字没换，就张冠李戴了。上面三条按分数从高到低排完是 1 Zh 95 / 2 Qi 82 / 3 Wu 70。

⑤结构体指针：p->成员

struct Stu *p = &a; 之后，p->score 就是 (*p).score，也就是 a.score。

括号不能省：*p.score 是 *(p.score)，因为 . 比 * 先算，而 p 不是结构体变量（细3）。

⑥结构体作参数：传值改不到，传指针才能改

传值
void raise_val(struct Stu s)
{
    s.score += 5;
}
// raise_val(a);
传指针
void raise_ptr(struct Stu *p)
{
    p->score += 5;
}
// raise_ptr(&a);

a.score 原来是 60：左边调用后是 60，右边是 65。和环09 ⑤ 的 swap 一模一样，这次复制过去的是整条记录。

结构体数组作参数时传的是地址，函数能直接改（环06 细6），所以题4 的 fun(STREC a[], int n) 排完序 main 里就看得到。

⑦链表：结点用 malloc 申请，next 串起来

struct Node {
    int num;
    struct Node *next;          // 指向下一个结点
};

三个结点串起来的样子：

head
 │
 ▼
┌────┬───┐   ┌────┬───┐   ┌────┬──────┐
│ 10 │ ●─┼──▶│ 20 │ ●─┼──▶│ 30 │ NULL │
└────┴───┘   └────┴───┘   └────┴──────┘

遍历：for (p = head; p != NULL; p = p->next) printf("%d ", p->num);

新结点用 q = (struct Node *)malloc(sizeof(struct Node)); 申请，记得判 NULL、把 q->next 设成 NULL。释放整条链表时，要先存下 next 再 free，不然 free 以后就找不到下一个了。

⑧删除结点：前一个的 next 跳过它

找到要删的结点 p，同时记住它前面那个 pre：

pre = NULL; p = head;
while (p != NULL && p->num != key) { pre = p; p = p->next; }
if (p != NULL) {
    if (pre == NULL) head = p->next;     // 删的是第一个：头指针要变
    else pre->next = p->next;            // 前一个直接指向后一个
    free(p);
}

10 → 20 → 30 删掉 20，剩下 10 → 30。

两个边界一定要想到：删的是第一个结点（head 要变，所以函数要返回新的 head）；找不到（p 走到 NULL，什么都不做）。

细6sizeof 结构体可能比成员加起来大

struct { char c; int x; } 的成员是 1 + 4 = 5 字节，sizeof 却是 8：编译器会补空位，让 int 落在 4 的倍数上。

考试一般问「至少多大」，或者按教材说「各成员之和」。具体补多少由编译器决定。

细7指向结构体数组的指针：p++ 跳一整条

struct Stu s[2] = {{1, "A", 50}, {2, "B", 60}}, *p = s; 执行 p++; 后 p->num 是 2：p 跳过了一整条记录，不是一个字节。道理和环11 细4 的指针相减一样，以「一个元素」为单位。

细8共用体 union：所有成员共用一块内存

union U { int i; char c; double d; } u;：

  • 大小至少是最大的成员：sizeof 是 8（double）。
  • 同一时刻只有最后写进去的那个有效：先 u.i = 65; 再 u.d = 1.5;，i 原来的值就被覆盖了。

和结构体对比：结构体每个成员各有各的格子，大小至少是成员之和。

细9枚举 enum：默认从 0 开始

enum color { RED, GREEN = 5, BLUE, BLACK };：RED = 0，GREEN = 5，BLUE = 6，BLACK = 7。

默认从 0 开始每个加 1；中间给了值，后面就从这个值接着往下加。枚举值是常量，不能被赋值。

细10typedef 起别名，和 #define 不一样

typedef struct Stu STU; 之后，STU s; 等于 struct Stu s;。typedef 不创造新类型，只是起个名字。

和 #define 的区别在指针上最明显：

typedef int *PINT; PINT a, b;→a、b 都是指针 #define PINT int * 后 PINT a, b;→只有 a 是指针 展开成 int *a, b;（环10 细1、环11 细1）
