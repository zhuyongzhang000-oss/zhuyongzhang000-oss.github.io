题1子串定位与替换

必做 · 2016 第6题（15 分）

编写一程序，判定一个字符串是否是另一个字符串的子串，若是，返回子串在主串中的位置。

要求：1、在主函数中输入两个字符串，调用子函数 cmpsubstr() 判断，并在主函数输出结果。

2、子函数的返回值为 -1 表示未找到，否则返回子串的位置（起始下标）。

要求

两个函数。

(1) cmpsubstr(s, t)：返回 t 第一次出现在 s 里的起始下标；找不到返回 -1。

t 是空串也返回 -1（学长参考程序的约定，考试时在注释里写明你用哪种约定）。

(2) replace_all(s, old, rep, out)：把 s 里所有的 old 从左往右、不重叠地换成 rep，结果放进 out。

old 是空串时原样复制。例如 "aaaa" 里的 "aa" 换成 "b" 得 "bb"；"aaa" 得 "ba"。

  • 依据2016⑥ 原题接口；学长清单「字符串匹配（暴力、kmp），很爱考字符串模拟题」。
  • 思路(1) 暴力匹配：每个起点 i 用 j 往后比；失配时起点只往后挪一格，不能直接跳到失配的位置。(2) 读游标 i 扫 s：从 i 起能匹配 old，就把 rep 抄进 out、i 跳过 old 的长度；否则抄一个字符、i++。

对应讲义　① ②

易错失配后起点只挪一格（"aab" 里找 "ab"）；替换时 i 跳过 old 的长度，不重叠；out 最后补 '\0'。

#include <stdio.h>
#include <string.h>

int cmpsubstr(const char *s, const char *t)
{
    int n = strlen(s), m = strlen(t), i, j;
    if (m == 0) return -1;                      /* 约定：空串算没找到 */
    for (i = 0; i + m <= n; i++) {              /* 起点最多到 n - m */
        for (j = 0; j < m && s[i + j] == t[j]; j++)
            ;
        if (j == m) return i;
    }
    return -1;
}

void replace_all(const char *s, const char *old, const char *rep, char *out)
{
    int m = strlen(old), i = 0, k = 0, j;
    while (s[i] != '\0') {
        if (m > 0 && strncmp(s + i, old, m) == 0) {
            for (j = 0; rep[j] != '\0'; j++) out[k++] = rep[j];
            i += m;                             /* 跳过整个 old：不重叠 */
        } else {
            out[k++] = s[i++];
        }
    }
    out[k] = '\0';
}

int main(void)
{
    char s[200], t[100], rep[100], out[400];
    int pos;
    fgets(s, sizeof s, stdin);              /* 主串可能有空格，用 fgets 读一整行 */
    s[strcspn(s, "\n")] = '\0';             /* 去掉 fgets 读进来的换行 */
    scanf("%99s %99s", t, rep);             /* 子串、换成什么 */
    pos = cmpsubstr(s, t);
    if (pos != -1) printf("子串在主串中的起始位置是 %d\n", pos);
    else printf("不是子串\n");
    replace_all(s, t, rep, out);
    printf("替换后：%s\n", out);
    return 0;
}

输入

she sells sea shells
se SE

运行结果

子串在主串中的起始位置是 4
替换后：she SElls SEa shells

题2单词逆序

必做 · 2017 第5题的延伸

输入一行英文，把单词的顺序颠倒后输出；单词之间只留一个空格，开头和结尾不留空格。

例如 "  I am   a student " 变成 "student a am I"。

要求

reverse_words(s, out)：结果放进 out。s 是空串或者全是空格时，out 是空串。s 不能改。

  • 依据2017⑤ 的约定「串中的词由 1 个或 1 个以上的空格符分隔」；学长「很爱考字符串模拟题」。
  • 思路从 s 的末尾往前扫：跳过空格 → 记下词尾 → 往前走到词头 → 把这个词抄进 out；out 里已经有词时，先补一个空格再抄。

对应讲义　③ ④

易错先跳空格再找词；第一个词前不加空格；全是空格时 out 也要是 ""。

#include <stdio.h>
#include <string.h>

void reverse_words(const char *s, char *out)
{
    int i = 0, k = 0, end, start, j;
    while (s[i] != '\0') i++;                   /* i 停在 '\0' */
    i--;
    while (i >= 0) {
        while (i >= 0 && s[i] == ' ') i--;      /* 跳过空格 */
        if (i < 0) break;
        end = i;                                /* 词尾 */
        while (i >= 0 && s[i] != ' ') i--;
        start = i + 1;                          /* 词头 */
        if (k > 0) out[k++] = ' ';
        for (j = start; j <= end; j++) out[k++] = s[j];
    }
    out[k] = '\0';
}

int main(void)
{
    char s[200], out[200];
    fgets(s, sizeof s, stdin);
    s[strcspn(s, "\n")] = '\0';
    reverse_words(s, out);
    printf("[%s]\n", out);
    return 0;
}

输入

  I am   a student

运行结果

[student a am I]

题3同数异形体

必做 · 2022 第2题（题干缺损，两种问法都出）

“同数异形体”的定义：由相同的数字（包括数字个数）、不同的排列形成的正整数，互为“同数异形体”。

例如：66167 和 16667 均由 3 个 6、1 个 1、1 个 7 组成，故互为同数异形体；而 6617 与 66671（数字个数不同）则不是。

要求

两个函数。

(1) same_form(a, b)：正整数 a、b 互为同数异形体返回 1，否则返回 0。

a == b 返回 0（定义里说的是「不同的排列」）。

(2) min_max_form(x, &mn, &mx)：在和 x 数字相同（个数也相同）的正整数里，找最小的和最大的（包括 x 自己），不能有前导 0。x 最多 9 位。

例如 x = 66167 → mn = 16667，mx = 76661；x = 1020 → mn = 1002，mx = 2100。

  • 依据2022②。
  • 注意原题转写到定义就断了，没写要求做什么。学长参考程序是「输入两个数，输出是否互为同数异形体」，这是最可能的问法（问法一）；问法二是同一个知识点（数字计数）的常见延伸。
  • 思路(1) 两个 cnt[10] 计数数组，各拆一遍位再逐个比较。不用排序。(2) 先计数。最大：从 9 往 0 依次放；最小：第一位放最小的非 0 数字，后面从 0 往 9 放。

对应讲义　⑤

易错计数数组比较代替排序；a == b 按定义返回 0；最小数的第一位不能是 0。

#include <stdio.h>

static void count_digits(int x, int cnt[10])
{
    int i;
    for (i = 0; i < 10; i++) cnt[i] = 0;
    do {                                        /* do-while：x 是 0 时也数一位 */
        cnt[x % 10]++;
        x /= 10;
    } while (x > 0);
}

int same_form(int a, int b)
{
    int ca[10], cb[10], i;
    if (a == b) return 0;
    count_digits(a, ca);
    count_digits(b, cb);
    for (i = 0; i < 10; i++)
        if (ca[i] != cb[i]) return 0;
    return 1;
}

void min_max_form(int x, int *mn, int *mx)
{
    int cnt[10], c[10], d, i, v;
    count_digits(x, cnt);
    v = 0;                                      /* 最大：9 到 0 */
    for (d = 9; d >= 0; d--)
        for (i = 0; i < cnt[d]; i++) v = v * 10 + d;
    *mx = v;
    for (d = 0; d < 10; d++) c[d] = cnt[d];
    for (d = 1; d < 10 && c[d] == 0; d++)       /* 最小的非 0 数字放第一位 */
        ;
    v = d;
    c[d]--;
    for (d = 0; d < 10; d++)
        for (i = 0; i < c[d]; i++) v = v * 10 + d;
    *mn = v;
}

int main(void)
{
    int a, b, mn, mx;
    scanf("%d %d", &a, &b);
    if (same_form(a, b)) printf("%d 和 %d 互为同数异形体\n", a, b);
    else printf("%d 和 %d 不是同数异形体\n", a, b);
    min_max_form(a, &mn, &mx);
    printf("和 %d 数字相同的数里，最小 %d，最大 %d\n", a, mn, mx);
    return 0;
}

输入

66167 16667

运行结果

66167 和 16667 互为同数异形体
和 66167 数字相同的数里，最小 16667，最大 76661

题4奇偶交替重构

选做 · 2021 第2题（题干缺损，两种理解都出）

输入一个正整数（如 2234），请将其各位数字重新构造（或逐位修正），生成一个新的正整数，使得其各位数字奇偶交替，并输出该新数。

要求

两个函数（x 最多 9 位）。

(1) fix_alternate(x)：逐位修正。最高位不动；从左往右，某一位和它左边那位（已经修正过的）奇偶相同，就把它改成 1（这里需要奇数）或 0（这里需要偶数）。2234 → 2101，1357 → 1050。

(2) rearrange_alternate(x)：重新排列。只用 x 原有的那些数字（个数不变），排成奇偶交替、没有前导 0 的最小的数；排不出来返回 -1。

3021 → 1032，2234 → -1（三个偶数一个奇数），100 → -1（只能是 0 开头）。

  • 依据2021②。
  • 注意「重新构造」和「逐位修正」是转写留下的两种说法，原题要哪种已经无法确定。考场上碰到题意不清，先在答卷上写一句「本题理解为……」，再按这个理解写。学长参考程序用的是逐位修正。
  • 思路(1) 先把各位拆进数组（高位在前），再从第 2 位起逐位看。(2) 奇数、偶数各有几个？差超过 1 就排不出。差 1 时开头的奇偶是定的；相等时两种开头都试，取小的。开头之后每一位的奇偶都定了，每次放剩下的同奇偶里最小的。

对应讲义　⑤ ⑥

易错(1) 和「修正后」的左边那位比；(2) 奇偶个数差 > 1 排不出，开头不能是 0，两种开头都要试。

#include <stdio.h>

static int split(int x, int d[])                /* 拆位，高位在前，返回位数 */
{
    int t[12], n = 0, i;
    do { t[n++] = x % 10; x /= 10; } while (x > 0);
    for (i = 0; i < n; i++) d[i] = t[n - 1 - i];
    return n;
}

int fix_alternate(int x)
{
    int d[12], n = split(x, d), i, v = d[0];
    for (i = 1; i < n; i++) {
        if (d[i] % 2 == d[i - 1] % 2)
            d[i] = d[i - 1] % 2 == 0 ? 1 : 0;   /* 左边偶 → 要奇数 → 1；左边奇 → 0 */
        v = v * 10 + d[i];
    }
    return v;
}

/* 第一位的奇偶是 first（1 奇 0 偶），按「剩下的同奇偶里最小的」排；排不出返回 -1 */
static int build(const int cnt0[10], int n, int first)
{
    int cnt[10], i, d, v = 0, want = first;
    for (i = 0; i < 10; i++) cnt[i] = cnt0[i];
    for (i = 0; i < n; i++) {
        for (d = want; d < 10; d += 2)
            if (cnt[d] > 0 && !(i == 0 && d == 0)) break;
        if (d >= 10) return -1;
        cnt[d]--;
        v = v * 10 + d;
        want = 1 - want;
    }
    return v;
}

int rearrange_alternate(int x)
{
    int d[12], n = split(x, d), cnt[10] = {0}, odd = 0, even, i, a, b;
    for (i = 0; i < n; i++) {
        cnt[d[i]]++;
        if (d[i] % 2) odd++;
    }
    even = n - odd;
    if (odd - even > 1 || even - odd > 1) return -1;
    if (odd > even) return build(cnt, n, 1);
    if (even > odd) return build(cnt, n, 0);
    a = build(cnt, n, 1);
    b = build(cnt, n, 0);
    if (a < 0) return b;
    if (b < 0) return a;
    return a < b ? a : b;
}

int main(void)
{
    int x, r;
    scanf("%d", &x);
    printf("逐位修正：%d\n", fix_alternate(x));
    r = rearrange_alternate(x);
    if (r == -1) printf("重新排列：排不出奇偶交替的数\n");
    else printf("重新排列：%d\n", r);
    return 0;
}

输入

2234

运行结果

逐位修正：2101
重新排列：排不出奇偶交替的数

套路

四道题用到的四个模板。每个模板先背骨架，再看它在哪一步最容易错。

暴力匹配骨架（题1）

for (i = 0; i + m <= n; i++) {          // 起点 i：最多到 n - m
    for (j = 0; j < m && s[i + j] == t[j]; j++)
        ;                               // 一直对上就往后比
    if (j == m) return i;               // 比完了 m 个：找到
}
return -1;

①暴力匹配：失配了起点只挪一格

外层是起点 i，内层是已经对上几个 j。失配时 j 清零、i 加 1，从下一个起点重新比。

最常见的错误是只用一个游标：s 走一步、t 对上就走一步、对不上 t 回到开头。"aab" 里找 "ab"：s[0] 对上 a，s[1] 是 a 不是 b，t 回到开头，但 s 已经走到 s[2] 了，于是 漏掉了从 s[1] 开始的那个 ab。

起点的上界：主串长 n、子串长 m，最后一个起点是 n − m。写成 i + m <= n 不用担心 m > n 时出负数。

"mississippi" 里找 "issip"→4 起点 1 时对上 "issi" 后失配，要回到起点 2 再试 "abc" 里找 ""→约定 strstr 认为在 0 找到；学长参考程序返回 −1。考试写明约定就行

KMP 能让失配时 i 不回退，学长清单里提到了。考研自命题卷写对暴力法就是满分，KMP 了解名字和作用即可，不用背代码。

②替换、删除：读游标 i、写游标 k 分开走

s 是读，out 是写，两个下标各管各的：

while (s[i] != '\0') {
    if (从 i 开始是 old) {              // strncmp(s + i, old, m) == 0
        把 rep 抄进 out，k 跟着走;
        i += m;                         // 跳过整个 old
    } else {
        out[k++] = s[i++];
    }
}
out[k] = '\0';
  • 「不重叠」就是匹配上以后 i 跳过整个 old。"aaa" 里 "aa" 换成 "b" 得 ba。
  • rep 里含有 old（"x" 换成 "xyz"）也不会死循环：写进 out 的东西不会再被读。在 s 上原地替换 才会出这种问题。
  • 删除就是 rep 为空串的替换；「删去所有空格」「删去所有数字字符」都是这个模板，条件换成判断一个字符。

③分词：跳空格 → 记词头 → 走到词尾

while (s[i] != '\0') {
    while (s[i] == ' ') i++;            // 1 跳过空格（可能有好几个）
    if (s[i] == '\0') break;            // 结尾的空格跳完就没词了
    start = i;                          // 2 词头
    while (s[i] != ' ' && s[i] != '\0') i++;
    len = i - start;                    // 3 词尾在 i - 1，长度 i - start
}

2017⑤ find_len_word、数单词个数、找最长的词、每个词首字母大写，都是这三步，只是第 3 步之后做的事不同。漏掉 if (s[i] == '\0') break; 的后果：串以空格结尾时 会多数出一个长度为 0 的词。

④单词逆序：倒着扫，或者两遍逆序

倒着扫（答案用的）：i 从最后一个字符往前，同样是「跳空格 → 记词尾 → 走到词头」，取到一个词就抄进 out。out 里已经有词，先补一个空格——这样空格只出现在词和词之间，开头结尾都不会多。

两遍逆序：先把整个串逆序 "I am" → "ma I"，再把每个词各自逆序 → am I。要改原串，适合「原地」的要求。

期末卷里的同类

从键盘输入两个字符串 a、b，连接成 c，再把 c 按升序排序输出 → 连接（环08 ⑤）+ 字符冒泡排序（环06 ①），排序的是 c[0] 到 c[len−1] 这些字符川师卷2016-17 四-2

⑤数字计数：cnt[10] 代替排序

「由相同的数字组成」= 0–9 每个数字出现的次数都相同。拆位时 cnt[x % 10]++，两个数各拆一遍，比较两个 cnt 数组。

do {
    cnt[x % 10]++;
    x /= 10;
} while (x > 0);                        // do-while：x 为 0 时也数一位

用计数数组构造数：

  • 最大：d 从 9 到 0，每个 d 放 cnt[d] 个。
  • 最小：第一位放最小的非 0 数字，其余 d 从 0 到 9 放。1020 的最小是 1002，不是 0012。

位数相同、数字不同的反例要会举：6617 和 66671 位数就不同；112 和 122 位数相同但数字个数不同。

⑥题意不清：先写理解，再按理解做

2021②、2022② 的题干都缺字。考场上碰到题意模糊：

  • 在答卷开头写一句「本题理解为：……」，再按这个理解写完整；
  • 选能写完、能自圆其说的理解，不要在两种理解之间来回改。

奇偶交替的「重排」理解，先算奇数、偶数各几个：差大于 1 一定排不出；差 1 时开头必须是多的那一类；相等时两种开头都试，取小的。开头还不能是 0：100 只能 0 开头，所以 排不出。

易错清单

写完对照一遍。

细1写进 out 的串，最后一定补 '\0'

测试会先把 out 填满 X。如果结果后面跟着一串 X，就是没补 '\0'。全是空格、空串这种「一个字符都没写」的情况也要补。

细2参数写成 const char *

const char *s 表示函数里不会通过 s 改字符串。传字符数组、字符串常量都可以。写 s[0] = 'x'; 编译器会报错，正好帮你检查「s 不能改」。

细3strncmp(s + i, old, m)：只比前 m 个

strncmp(a, b, m) 比较前 m 个字符，相同返回 0。strncmp("abcde" + 1, "bcx", 2) 的值是 0。不想用库函数，就写一个 j 循环逐个比（骨架那样）。
