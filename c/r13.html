题1转大写存文件

代表题 · 教材第10章习题3（也是圣才考研真题精选综合题7）

从键盘输入一个字符串，将其中的小写字母全部转换成大写字母，然后输出到一个磁盘文件 test 中保存，输入的字符串以「!」结束。

要求

save_upper(in, filename)：从 in 一个字符一个字符地读（fgetc(in)），读到 '!' 或文件结束就停；小写字母变大写，其他字符原样；全部写进名为 filename 的文件（"w" 方式），最后关闭文件。

'!' 本身不写。考试时把 fgetc(in) 换成 getchar() 就是原题。

对应讲义　① ③

易错ch 用 int；判 NULL；'!' 不写；最后 fclose。

#include <stdio.h>

void save_upper(FILE *in, const char *filename)
{
    FILE *fp = fopen(filename, "w");
    int ch;
    if (fp == NULL) { printf("打不开文件\n"); return; }
    while ((ch = fgetc(in)) != EOF && ch != '!') {
        if (ch >= 'a' && ch <= 'z') ch = ch - 32;
        fputc(ch, fp);
    }
    fclose(fp);
}

int main(void)
{
    save_upper(stdin, "test");       /* 键盘就是 stdin */
    return 0;
}

输入

Hello, world! 后面的不要

运行后文件 test 的内容

HELLO, WORLD

题2写入再读出

变式 · 教材第10章习题11

从键盘输入若干行字符（每行长度不等），输入后把它们存储到一磁盘文件中。

再从该文件中读入这些数据，将其中小写字母转换成大写字母后在显示屏上输出。

要求

写两个函数：

save_lines(in, filename)：把 in 里的全部内容（读到 EOF 为止）原样写进文件 filename。

show_upper(filename)：打开文件，把内容读出来，小写变大写后输出到屏幕（printf 或 putchar）。

用 fgetc/fputc 或 fgets/fputs 都可以。考试时 in 就是 stdin（键盘）。

对应讲义　③ ④

易错按字符复制；读出来时转大写再输出。

#include <stdio.h>

void save_lines(FILE *in, const char *filename)
{
    FILE *fp = fopen(filename, "w");
    int ch;
    if (fp == NULL) return;
    while ((ch = fgetc(in)) != EOF) fputc(ch, fp);
    fclose(fp);
}

void show_upper(const char *filename)
{
    FILE *fp = fopen(filename, "r");
    int ch;
    if (fp == NULL) return;
    while ((ch = fgetc(fp)) != EOF) {
        if (ch >= 'a' && ch <= 'z') ch = ch - 32;
        putchar(ch);
    }
    fclose(fp);
}

int main(void)
{
    save_lines(stdin, "text.txt");   /* 输入完按 Ctrl+Z（Windows）结束 */
    show_upper("text.txt");
    return 0;
}

File is easy.

题3成绩写文件

变式 · 教材第10章习题5

有 5 个学生，每个学生有 3 门课程的成绩，从键盘输入学生数据（包括学号、姓名、3 门课程成绩），计算出平均成绩，将原有数据和计算出的平均分数存放在磁盘文件 stud 中。

要求

save_students(s, n, filename)：先算出每个学生的 s[i].avg（三门平均），再把每个学生写成一行："学号 姓名 成绩1 成绩2 成绩3 平均\n"，成绩保留 1 位小数，平均保留 2 位，例如 "101 Li 80.0 90.0 70.0 80.00\n"。

对应讲义　⑤

易错先算平均再写；格式按题目要求。

#include <stdio.h>

struct Stud {
    int num;
    char name[20];
    double score[3];
    double avg;
};

void save_students(struct Stud s[], int n, const char *filename)
{
    FILE *fp = fopen(filename, "w");
    int i;
    if (fp == NULL) return;
    for (i = 0; i < n; i++) {
        s[i].avg = (s[i].score[0] + s[i].score[1] + s[i].score[2]) / 3;
        fprintf(fp, "%d %s %.1f %.1f %.1f %.2f\n", s[i].num, s[i].name,
                s[i].score[0], s[i].score[1], s[i].score[2], s[i].avg);
    }
    fclose(fp);
}

#define N 5

int main(void)
{
    struct Stud s[N];
    int i;
    for (i = 0; i < N; i++)
        scanf("%d%s%lf%lf%lf", &s[i].num, s[i].name, &s[i].score[0], &s[i].score[1], &s[i].score[2]);
    save_students(s, N, "stud");
    return 0;
}

题4两文件合并

综合 · 教材第10章习题4

有两个磁盘文件 A 和 B，各存放一行字母，要求把这两个文件中的信息合并（按字母顺序排列），输出到一个新文件 C 中去。

要求

merge_files(fa, fb, fc)：把文件 fa、fb 里的字母（换行符不要）都读进一个字符数组，从小到大排好，写进文件 fc，最后再写一个 '\n'。

  • 思路读 → 排序（环06 的冒泡）→ 写。数组开 200 就够。

对应讲义　③

易错读两个文件进同一个数组（跳过换行）→ 冒泡排序 → 写。

#include <stdio.h>

void merge_files(const char *fa, const char *fb, const char *fc)
{
    char c[200];
    int n = 0, ch, i, j, t;
    const char *names[2];
    FILE *fp;
    names[0] = fa;
    names[1] = fb;
    for (i = 0; i < 2; i++) {
        fp = fopen(names[i], "r");
        if (fp == NULL) return;
        while ((ch = fgetc(fp)) != EOF && n < 199)
            if (ch != '\n') c[n++] = (char)ch;
        fclose(fp);
    }
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - 1 - i; j++)
            if (c[j] > c[j + 1]) { t = c[j]; c[j] = c[j + 1]; c[j + 1] = (char)t; }
    fp = fopen(fc, "w");
    if (fp == NULL) return;
    for (i = 0; i < n; i++) fputc(c[i], fp);
    fputc('\n', fp);
    fclose(fp);
}

int main(void)
{
    merge_files("A", "B", "C");
    return 0;
}

文件 B 原有内容

运行后文件 C 的内容

题5文件排序

限时 · 教材第10章习题6 · 限时 20 分钟

将第 5 题 stud 文件中的学生数据，按平均分进行排序处理，将已排序的学生数据存入一个新文件 stu_sort 中。

要求

sort_file(in, out)：in 文件是题3 的格式（每行 "学号 姓名 成绩1 成绩2 成绩3 平均"），读出所有学生（最多 50 个，用 fscanf 读到读不进来为止），按平均分从高到低排序，再用同样的格式写进 out 文件。

对应讲义　⑤ 环12 ④

易错fscanf 返回 6 才算读到一条；整条记录交换；同样格式写回。

#include <stdio.h>

struct Stud {
    int num;
    char name[20];
    double score[3];
    double avg;
};

void sort_file(const char *in, const char *out)
{
    struct Stud s[50], t;
    int n = 0, i, j;
    FILE *fp = fopen(in, "r");
    if (fp == NULL) return;
    while (n < 50 && fscanf(fp, "%d %s %lf %lf %lf %lf", &s[n].num, s[n].name,
                            &s[n].score[0], &s[n].score[1], &s[n].score[2], &s[n].avg) == 6)
        n++;
    fclose(fp);
    for (i = 0; i < n - 1; i++)
        for (j = 0; j < n - 1 - i; j++)
            if (s[j].avg < s[j + 1].avg) { t = s[j]; s[j] = s[j + 1]; s[j + 1] = t; }
    fp = fopen(out, "w");
    if (fp == NULL) return;
    for (i = 0; i < n; i++)
        fprintf(fp, "%d %s %.1f %.1f %.1f %.2f\n", s[i].num, s[i].name,
                s[i].score[0], s[i].score[1], s[i].score[2], s[i].avg);
    fclose(fp);
}

int main(void)
{
    sort_file("stud", "stu_sort");
    return 0;
}

编程题怎么写

文件读写函数和屏幕输入输出一一对应，只是多了一个 fp。

通用骨架

FILE *fp = fopen("文件名", "r");          // 打开
if (fp == NULL) {                          // 判断
    printf("打开失败\n");
    return 1;
}
while ((ch = fgetc(fp)) != EOF) { … }      // 读写；ch 定义成 int
fclose(fp);                                // 关闭

①打开 → 判断 → 读写 → 关闭

每一步都不能少：

  • 不判 NULL：文件打不开时（文件不存在、路径写错）fp 是 NULL，接着读写程序就崩溃。
  • 不 fclose：写的内容可能还在缓冲区里，没真正写进文件（细8）。

②打开方式：r、w、a

"r"→只读 文件必须存在，不存在返回 NULL "w"→只写 不存在就新建；存在就先清空 "a"→追加 写在原来内容的后面 "r+" / "w+" / "a+"→读写都行 各自保留上面的特点（w+ 也会先清空） "rb"、"wb"……→二进制方式 不加 b 是文本方式

想在文件后面接着写却用了 "w"，原来的内容就没了。

真题里这样考

fp = fopen(filename, "rb+");（filename 里存的是 "paper"）=> 以读写方式打开二进制文件 paper真选29·华南理工2018

填空考 fopen 的打开方式川师卷

③按字符读写：fgetc、fputc

fgetc(fp) 读一个字符，读到文件尾返回 EOF（值是 −1）；fputc(ch, fp) 写一个字符。

while ((ch = fgetc(fp)) != EOF) {     // 括号一个都不能少（回顾第 2 题）
    fputc(toupper(ch), out);
}

ch 要定义成 int：EOF 是 −1，用 char 接，有的编译器上永远等不到 EOF（细6）。

一个写了 hello file⏎second line⏎ 的文件，用这个循环数字符，一共 23 个（两个换行也算）。

④按行读写：fgets 会把回车也读进来

fgets(line, 大小, fp) 读一行，行末的 '\n' 也在 line 里；读到文件尾返回 NULL。fputs(s, fp) 写一个字符串，不会自动加换行。

while (fgets(line, sizeof line, fp) != NULL)
    printf("[%s]", line);

上面那个文件读出来是 [hello file⏎]：方括号里能看到回车也被读进来了。要去掉它，就把最后那个 '\n' 改成 '\0'（环08 ①）。

⑤按格式读写：fprintf、fscanf

和 printf、scanf 用法一样，只是第一个参数多一个 fp：

fprintf(fp, "%d %s %.1f\n", s.num, s.name, s.score);

while (fscanf(fp, "%d %s %lf", &num, name, &score) == 3)
    …

fscanf 返回成功读到几项。这里要读 3 项，所以写 == 3；读到文件尾时返回的不是 3，循环就停了。

题3、题5 都用这一对：fprintf 一行写一个学生，fscanf 一行读一个学生。

⑥按块读写：fread、fwrite

fwrite(地址, 每块多大, 写几块, fp);
fread (地址, 每块多大, 读几块, fp);   // 返回成功读到的块数

把结构体数组原样存进文件、再原样读回来：fwrite(stu, sizeof(struct Stud), 5, fp);。要用二进制方式打开（"wb"、"rb"）。

教材答案多用这一对写学生数据。这个仓库的题3 改用 fprintf 写文本，是为了测试能逐字检查。

⑦while (!feof(fp)) 会多处理一次

feof 要等读失败过一次才变成真。所以：

while (!feof(fp)) {
    ch = fgetc(fp);
    n++;
}

文件里只有 ab 两个字符，这个循环却跑了 3 次：最后一次 fgetc 读到的是 EOF，n 也加了。

正确写法是「先读，再判断」：while ((ch = fgetc(fp)) != EOF)。

⑧文件位置：rewind、fseek、ftell

  • rewind(fp)：回到文件开头。
  • fseek(fp, 偏移, 起点)：移动位置。起点 SEEK_SET（开头，0）、SEEK_CUR（当前，1）、SEEK_END（末尾，2）。
  • ftell(fp)：当前位置离开头几个字节。

fseek(fp, 0, SEEK_END); 之后 ftell(fp) 就是文件大小：ab 这个文件是 2 字节。

文件的概念题

每一节最后附了真题的问法，灰色小字是出处。

细1FILE 是一个结构体类型

FILE 是 stdio.h 里定义的结构体类型，大写。程序里定义的是指针 FILE *fp;。file *fp、FILE fp 都不对。

细2文本方式和二进制方式

文本方式在 Windows 上会转换换行（'\n' 和回车换行互转）；二进制方式原样读写。

写给人看的（成绩单）用文本 + fprintf；写结构体、数组这种「内存原样」用二进制 + fwrite。

细3"w" 会先清空文件

打开已有文件用 "w"，原来的内容立刻没了；想接在后面写用 "a"（②）。

细4feof(fp) 的返回值

已经读到文件尾（读失败过）返回非 0；还没结束返回 0。

真题里这样考

文件没有结束时 feof 的返回值 → 0真选28·华南理工2018

细5fopen 的第一个参数：文件名字符串

可以是常量 "paper"，也可以是存着文件名的字符数组：char filename[] = "paper"; fopen(filename, "r") 打开的是 paper，不是一个叫 filename 的文件。

路径里的反斜杠要写两个："c:\\test\\out.dat"（环02 细4 的转义）。

细6EOF 是 −1，所以 ch 是 int

EOF 的值是 -1。用 char ch 接 fgetc 的返回值，有的编译器上 char 存不下 −1，永远等不到 EOF；有的遇到 0xFF 这个字节会提前停。所以 int ch;。

细7stdin、stdout 也是 FILE *

键盘和屏幕也被当成文件：fgetc(stdin) 就是 getchar()，fprintf(stdout, …) 就是 printf(…)。

所以「从键盘输入写进文件」的题，读键盘可以写成 fgetc(stdin)，和读文件是一样的代码。仓库里的题1、题2 就是把 in 当参数传进去，测试时 in 是一个文件，考试时就是 stdin。

细8打开过的文件要 fclose

写文件时数据先放在缓冲区，攒够了才真正写进去。不 fclose，程序结束时最后那点数据可能还没写进文件。fclose 会把缓冲区的内容写完再关。
