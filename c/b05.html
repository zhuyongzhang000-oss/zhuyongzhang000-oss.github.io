题1建表和有序插入

必做 · 2018 第5题搬到链表上

用尾插法把数组建成链表；再往一个升序链表里插入一个数，插入后仍然升序。

要求

两个函数（结点类型见 node.h，链表不带头结点）。

(1) build(a, n)：按 a[0] ~ a[n-1] 的顺序建链表（尾插法），返回头指针；n == 0 返回 NULL。

(2) insert_sorted(head, x)：head 是升序链表（可以是空的），新建一个结点放 x 插进去，插在第一个比 x 大的结点前面（和 x 相等的已有结点留在它前面），返回新的头指针。

  • 依据2018⑤「有序插入（函数）」（20 分，数组版在基础环09 题4）；教材 9.4 建立动态链表。单考没有数据结构卷了，第 9 章链表很可能直接进 C 卷（推测）。
  • 思路(1) head、tail 两个指针，新结点接在 tail 后面；第一个结点要单独让 head 指向它。(2) 插在最前面（空表、或者 x 比第一个还小）要改 head，单独处理；其余情况 p 停在「插入位置的前一个」：while (p->next != NULL && p->next->data <= x) p = p->next;

对应讲义　① ②

易错第一个结点要让 head 指向它；插在最前面要改 head；p 停在插入位置的前一个。

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *build(const int a[], int n)
{
    struct node *head = NULL, *tail = NULL, *q;
    int i;
    for (i = 0; i < n; i++) {
        q = (struct node *)malloc(sizeof(struct node));
        q->data = a[i];
        q->next = NULL;
        if (head == NULL) head = q;
        else tail->next = q;
        tail = q;
    }
    return head;
}

struct node *insert_sorted(struct node *head, int x)
{
    struct node *q = (struct node *)malloc(sizeof(struct node)), *p;
    q->data = x;
    if (head == NULL || x < head->data) {           /* 插在最前面 */
        q->next = head;
        return q;
    }
    p = head;
    while (p->next != NULL && p->next->data <= x) p = p->next;
    q->next = p->next;                              /* 先接后面，再接前面 */
    p->next = q;
    return head;
}

void print_list(struct node *head)
{
    struct node *p;
    for (p = head; p != NULL; p = p->next) printf("%d ", p->data);
    printf("\n");
}

int main(void)
{
    int a[100], n, i, x;
    struct node *head;
    scanf("%d", &n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    head = build(a, n);
    print_list(head);
    scanf("%d", &x);
    head = insert_sorted(head, x);
    print_list(head);
    return 0;
}

输入

5
1 3 5 7 9
6

运行结果

1 3 5 7 9 
1 3 5 6 7 9

题2逆置和删除

必做 · 教材第 9 章

链表的两个基本改造：原地逆置；删除所有值等于 x 的结点。

要求

两个函数（结点类型见 node.h，链表不带头结点）。

(1) reverse(head)：把链表原地逆置（不新建结点，只改 next），返回新的头指针。空链表返回 NULL。

(2) remove_all(head, x)：删除所有 data == x 的结点并 free，返回新的头指针。

连着好几个、开头就是、全部都是，都要处理。

  • 依据教材第 9 章链表；基础环12 题5 只删第一个匹配的结点，这里要删全部、还要逆置。
  • 思路(1) 三个指针 pre、p、nxt：先存 nxt = p->next，再让 p->next = pre，然后 pre、p 都往前走一步。(2) 先用 while 把开头的 x 都删掉（每次 head 往后移）；再让 p 从 head 开始，看 p->next 要不要删。

对应讲义　③ ④

易错逆置先存 nxt 再改 p->next；删除时开头的要循环删，中间删完 p 不要往前走。

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *reverse(struct node *head)
{
    struct node *pre = NULL, *p = head, *nxt;
    while (p != NULL) {
        nxt = p->next;
        p->next = pre;
        pre = p;
        p = nxt;
    }
    return pre;
}

struct node *remove_all(struct node *head, int x)
{
    struct node *p, *q;
    while (head != NULL && head->data == x) {       /* 开头的 x 全删掉 */
        q = head;
        head = head->next;
        free(q);
    }
    p = head;
    while (p != NULL && p->next != NULL) {
        if (p->next->data == x) {
            q = p->next;
            p->next = q->next;                      /* 删掉以后 p 不动，再看新的 p->next */
            free(q);
        } else {
            p = p->next;
        }
    }
    return head;
}

struct node *creat(void)              /* 读到 0 结束，尾插法 */
{
    struct node *head = NULL, *tail = NULL, *p;
    int x;
    scanf("%d", &x);
    while (x != 0) {
        p = (struct node *)malloc(sizeof(struct node));
        p->data = x;
        p->next = NULL;
        if (head == NULL) head = p;
        else tail->next = p;
        tail = p;
        scanf("%d", &x);
    }
    return head;
}

void print_list(struct node *head)
{
    struct node *p;
    for (p = head; p != NULL; p = p->next) printf("%d ", p->data);
    printf("\n");
}

int main(void)
{
    struct node *head = creat();
    int x;
    head = reverse(head);
    print_list(head);
    scanf("%d", &x);
    head = remove_all(head, x);
    print_list(head);
    return 0;
}

输入

2 2 1 2 3 2 2 0
2

运行结果

2 2 3 2 1 2 2 
3 1

题3有序合并去重

必做 · 2015 第5题搬到链表上

a、b 是两个升序链表（各自内部也可能有重复的数），把它们合并成一个严格升序的链表：

相同的值只留一个。

要求

merge_unique(a, b)：不新建结点，直接把原来的结点重新串起来；多出来的（重复的）结点 free 掉。

返回合并后的头指针。a、b 都可以是空链表。

例如 a = 1 3 3 5，b = 2 3 6 → 1 2 3 5 6。

  • 依据2015⑤「两数组排序后合并成一个升序数组」（20 分，数组版在基础环06 题5）。
  • 思路用一个临时的「哨兵」结点 struct node dummy; tail = &dummy; 每次从 a、b 的头上摘下较小的那个：它和 tail 的值相同（并且 tail 不是哨兵）就 free，否则接到 tail 后面。最后 tail->next = NULL，返回 dummy.next。

对应讲义　⑤

易错哨兵结点省掉「第一个」的特判；和 tail 相同就 free；最后 tail->next = NULL。

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

struct node *merge_unique(struct node *a, struct node *b)
{
    struct node dummy, *tail = &dummy, *q;
    while (a != NULL || b != NULL) {
        if (b == NULL || (a != NULL && a->data <= b->data)) {
            q = a; a = a->next;                     /* 从 a 摘下第一个 */
        } else {
            q = b; b = b->next;
        }
        if (tail != &dummy && tail->data == q->data) {
            free(q);                                /* 重复的 */
        } else {
            tail->next = q;
            tail = q;
        }
    }
    tail->next = NULL;
    return dummy.next;
}

struct node *creat(void)              /* 读到 0 结束，尾插法 */
{
    struct node *head = NULL, *tail = NULL, *p;
    int x;
    scanf("%d", &x);
    while (x != 0) {
        p = (struct node *)malloc(sizeof(struct node));
        p->data = x;
        p->next = NULL;
        if (head == NULL) head = p;
        else tail->next = p;
        tail = p;
        scanf("%d", &x);
    }
    return head;
}

int main(void)
{
    struct node *a = creat(), *b = creat(), *p;
    for (p = merge_unique(a, b); p != NULL; p = p->next) printf("%d ", p->data);
    printf("\n");
    return 0;
}

输入

1 3 3 5 0
2 3 6 0

运行结果

1 2 3 5 6

题4约瑟夫环

选做 · 川师本科期末卷 2018-19 五-2（15 分）

编号为 1~n 的 n 个人围成一圈，假设从第 2 个人开始报数（从 1 到 3 报数），凡数到 3 的人退出圈子，然后下一个人又从 1 到 3 报数，数到 3 的人退出圈子，直到所有人都退出圈子，编程输出这 n 个人退出的序号。

例如，n=5，退出圈子的序号为：42135。

要求

josephus(n, start, m, out)：从编号 start 的人开始报数，数到 m 的人退出；按退出的先后把编号依次放进 out[0] ~ out[n-1]。1 <= start <= n，m >= 1。

原题就是 josephus(5, 2, 3, out)，out = {4, 2, 1, 3, 5}。

要求用循环链表（最后一个结点的 next 指回第一个），函数结束前把结点都 free 掉。

  • 依据期末卷原题。
  • 思路建好 1 ~ n 的循环链表后，让 pre 指向「开始报数的人」的前一个结点；每轮 pre 往前走 m - 1 步，pre->next 就是数到 m 的人：记下编号、从环里摘掉、free。

对应讲义　⑥

易错尾结点接回第一个成环；pre 停在报数起点的前一个；每轮走 m-1 步，摘掉 pre->next。

#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *next;
};

void josephus(int n, int start, int m, int out[])
{
    struct node *head = NULL, *tail = NULL, *q, *pre;
    int i, k = 0;
    for (i = 1; i <= n; i++) {                      /* 建 1 ~ n */
        q = (struct node *)malloc(sizeof(struct node));
        q->data = i;
        if (head == NULL) head = q;
        else tail->next = q;
        tail = q;
    }
    tail->next = head;                              /* 成环 */
    pre = tail;                                     /* pre 是第 1 个人的前一个 */
    for (i = 1; i < start; i++) pre = pre->next;    /* 挪到 start 的前一个 */
    while (k < n) {
        for (i = 1; i < m; i++) pre = pre->next;    /* 数 m-1 个 */
        q = pre->next;                              /* 数到 m 的人 */
        out[k++] = q->data;
        pre->next = q->next;
        free(q);
    }
}

int main(void)
{
    int n, out[100], i;
    scanf("%d", &n);
    josephus(n, 2, 3, out);            /* 原题：从第 2 个人开始，数到 3 退出 */
    for (i = 0; i < n; i++) printf("%d", out[i]);
    printf("\n");
    return 0;
}

输入

5

运行结果

42135

开始之前，先回想一下

这些是前面学过、这一环又要用到的基础。想不起来就回到出处那一节再看一眼。

p->data 等于 → (*p).data环12 细3

malloc 来的结点用完要 → free环11 细10

删除结点时删的是第一个，要 → 改 head环12 ⑧

有序数组插入要找 → 第一个比 x 大的位置环06 ⑤

两个有序数组合并，一边走完要 → 把另一边剩下的接上环06 ⑥

套路

先画图，再按图改指针。

遍历骨架

struct node *p;
for (p = head; p != NULL; p = p->next)
    printf("%d ", p->data);

这一环的链表都不带头结点：head 直接指向第一个数据结点，空链表 head == NULL。所以凡是可能改到第一个结点的操作（插在最前、删第一个、逆置），函数都要返回新的 head，调用时写 head = f(head, …);。

①尾插法：head 和 tail 两个指针

q = (struct node *)malloc(sizeof(struct node));
q->data = a[i];
q->next = NULL;                 // 新结点永远是最后一个
if (head == NULL) head = q;     // 第一个结点：让 head 指向它
else tail->next = q;            // 否则接在 tail 后面
tail = q;

头插法（q->next = head; head = q;）更短，但建出来的顺序是反的。题目要「按输入顺序」就用尾插。

②插入：p 停在插入位置的前一个

单链表只能往后走，要在某个结点前面插，p 必须停在它的前一个：

if (head == NULL || x < head->data) {     // 插在最前面：改 head
    q->next = head;
    return q;
}
p = head;
while (p->next != NULL && p->next->data <= x) p = p->next;
q->next = p->next;                        // ① 先接后面
p->next = q;                              // ② 再接前面

两句顺序反了：先写 p->next = q;，p 后面原来那一段就找不到了。

<= x 让相等的已有结点留在前面；写成 < x 新结点就插在相等的前面。题目没规定时两种都对。

③原地逆置：pre、p、nxt 三个指针

pre = NULL;  p = head;
while (p != NULL) {
    nxt = p->next;      // 1 先记住后面
    p->next = pre;      // 2 掉头
    pre = p;            // 3 两个都往前走
    p = nxt;
}
return pre;             // pre 停在原来的最后一个，就是新的 head

循环结束时 p 是 NULL，所以返回的是 pre。空链表、一个结点代进去都对，不用特判。

④删除所有等于 x 的：开头单独删，中间看 p->next

while (head != NULL && head->data == x) {     // 开头可能连着好几个
    q = head;  head = head->next;  free(q);
}
p = head;
while (p != NULL && p->next != NULL) {
    if (p->next->data == x) {
        q = p->next;  p->next = q->next;  free(q);   // 删完 p 不动
    } else
        p = p->next;
}

删完一个以后 p 不能往前走：新的 p->next 也可能是 x。2 2 1 2 3 2 2 删 2 得 1 3。

free(q) 之后不能再用 q：先 p->next = q->next; 再 free，顺序不能反。

⑤合并：哨兵结点省掉「第一个」的特判

在栈上放一个不存数据的结点 dummy，tail 从它开始接，最后返回 dummy.next：

struct node dummy, *tail = &dummy;
while (a != NULL || b != NULL) {
    if (b == NULL || (a != NULL && a->data <= b->data)) { q = a; a = a->next; }
    else { q = b; b = b->next; }
    tail->next = q;  tail = q;      // 接上（去重时先和 tail 比）
}
tail->next = NULL;
return dummy.next;

有了哨兵，「结果链表是空的」和「不是空的」写法一样。去重：q 的值和 tail 相同（并且 tail 不是 &dummy）就 free(q)，不接。最后的 tail->next = NULL; 不能漏：最后接上的结点原来的 next 可能还指着已经 free 的结点。

⑥循环链表：约瑟夫环

建好以后 tail->next = head; 就成环了。环上没有 NULL，遍历的结束条件改成「回到出发点」或者「数够了个数」。

约瑟夫环：pre 指向「下一个开始报数的人」的前一个，每轮：

for (i = 1; i < m; i++) pre = pre->next;   // 走 m - 1 步
q = pre->next;                             // 数到 m 的人
out[k++] = q->data;
pre->next = q->next;                       // 摘掉
free(q);

摘掉以后 pre 不动，pre->next 就是下一个开始报数的人。从第 2 个人开始、数到 3，5 个人的退出顺序是 4 2 1 3 5。

期末卷原题

n 个人围成一圈，从第 2 个人开始 1 到 3 报数，数到 3 的退出，输出退出的序号；n = 5 时为 42135 → 循环链表（本节），也可以用数组 + 标记做（环06 的数组）川师卷2018-19 五-2

静态链表中的指针表示的是 → 数组下标川师卷2022-23 一-8

易错清单

写完对照一遍。

细1先判断 p != NULL，再写 p->next

while (p->next != NULL && …) 的前提是 p 本身不是 NULL。空链表直接进这个循环会崩溃。&& 的短路（环03 细3）让 p != NULL && p->data == x 这样写是安全的，两边顺序反了就不安全。

细2结点必须 malloc，不能用局部变量

struct node t;  t.data = x;  p->next = &t;   // 错：函数结束 t 就没了

只有哨兵 dummy 这种「只在函数里用、不放进结果」的结点可以是局部变量。

细3malloc(sizeof(struct node))

malloc(sizeof(struct node *)) 只分配了一个指针的大小。malloc 返回 void *，C 里可以不强转，教材习惯写 (struct node *)。
